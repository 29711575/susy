if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  # MKLROOT is a cache identity.  Normalize it before looking up any
  # component so an equivalent symlink does not create a second identity.
  # An explicit -DMKLROOT (or an existing cache choice) takes precedence over
  # the environment; MKLROOT in the environment only initializes an unset
  # cache.
  set(_mkl_requested_root "")
  set(_mkl_selection "auto")
  if(DEFINED MKLROOT AND NOT "${MKLROOT}" STREQUAL "")
    set(_mkl_requested_root "${MKLROOT}")
    set(_mkl_selection "cache-or-explicit")
  elseif(DEFINED ENV{MKLROOT} AND NOT "$ENV{MKLROOT}" STREQUAL "")
    set(_mkl_requested_root "$ENV{MKLROOT}")
    set(_mkl_selection "environment")
  endif()

  if(_mkl_requested_root)
    get_filename_component(_mkl_root_absolute "${_mkl_requested_root}" ABSOLUTE)
    if(EXISTS "${_mkl_root_absolute}")
      get_filename_component(_mkl_root "${_mkl_root_absolute}" REALPATH)
    else()
      # Keep an invalid requested root visible in the diagnostic.  In
      # particular, do not let a previous library cache make it look valid.
      set(_mkl_root "${_mkl_root_absolute}")
    endif()
  else()
    file(GLOB _mkl_root_candidates LIST_DIRECTORIES true
      "/opt/intel/oneapi/mkl/*")
    list(SORT _mkl_root_candidates)
    list(REVERSE _mkl_root_candidates)
    foreach(_mkl_candidate IN LISTS _mkl_root_candidates)
      if(EXISTS "${_mkl_candidate}/include/mkl_types.h")
        get_filename_component(_mkl_root "${_mkl_candidate}" REALPATH)
        break()
      endif()
    endforeach()
  endif()
  if(NOT _mkl_root)
    message(FATAL_ERROR
      "MKL root not found. Set MKLROOT to the oneAPI MKL root directory.")
  endif()

  if(DEFINED MKL_RESOLVED_ROOT AND
     NOT "${MKL_RESOLVED_ROOT}" STREQUAL "${_mkl_root}")
    message(STATUS "MKL root changed: clearing cached MKL component paths")
    unset(MKL_INCLUDE_DIR CACHE)
    foreach(_mkl_component intel_lp64 sequential core)
      string(TOUPPER "${_mkl_component}" _mkl_component_upper)
      unset(MKL_${_mkl_component_upper}_LIBRARY CACHE)
    endforeach()
  endif()
  set(MKLROOT "${_mkl_root}" CACHE PATH "Intel oneAPI MKL root directory" FORCE)
  set(MKL_RESOLVED_ROOT "${_mkl_root}" CACHE INTERNAL
    "Normalized MKL root used for component discovery" FORCE)
  set(MKL_INCLUDE_DIR "${_mkl_root}/include" CACHE PATH
    "Intel oneAPI MKL include directory" FORCE)
  if(NOT EXISTS "${MKL_INCLUDE_DIR}/mkl_types.h")
    message(FATAL_ERROR
      "MKL headers not found below requested root ${_mkl_root}.")
  endif()

  # Reject a stale or user-injected component path before find_library can
  # honor it from the cache.  The real path must remain below this root.
  function(_mkl_path_below_root path root output)
    if(NOT EXISTS "${path}")
      set(${output} FALSE PARENT_SCOPE)
      return()
    endif()
    get_filename_component(_path_real "${path}" REALPATH)
    get_filename_component(_root_real "${root}" REALPATH)
    string(LENGTH "${_root_real}" _root_length)
    string(LENGTH "${_path_real}" _path_length)
    set(_inside FALSE)
    if(_path_length GREATER_EQUAL _root_length)
      string(SUBSTRING "${_path_real}" 0 ${_root_length} _prefix)
      if("${_prefix}" STREQUAL "${_root_real}")
        if(_path_length EQUAL _root_length)
          set(_inside TRUE)
        else()
          string(SUBSTRING "${_path_real}" ${_root_length} 1 _separator)
          if("${_separator}" STREQUAL "/")
            set(_inside TRUE)
          endif()
        endif()
      endif()
    endif()
    set(${output} ${_inside} PARENT_SCOPE)
  endfunction()

  get_filename_component(_mkl_include_real "${MKL_INCLUDE_DIR}" REALPATH)
  _mkl_path_below_root("${_mkl_include_real}" "${_mkl_root}" _mkl_include_inside)
  if(NOT _mkl_include_inside)
    message(FATAL_ERROR
      "MKL include directory is outside requested root: ${MKL_INCLUDE_DIR}")
  endif()
  foreach(_mkl_component intel_lp64 sequential core)
    string(TOUPPER "${_mkl_component}" _mkl_component_upper)
    if(DEFINED MKL_${_mkl_component_upper}_LIBRARY)
      _mkl_path_below_root("${MKL_${_mkl_component_upper}_LIBRARY}" "${_mkl_root}" _mkl_cached_inside)
      if(NOT _mkl_cached_inside)
        unset(MKL_${_mkl_component_upper}_LIBRARY CACHE)
        unset(MKL_${_mkl_component_upper}_LIBRARY)
      endif()
    endif()
    find_library(MKL_${_mkl_component_upper}_LIBRARY NAMES "mkl_${_mkl_component}"
      HINTS "${MKLROOT}/lib" "${MKLROOT}/lib/intel64"
      NO_DEFAULT_PATH)
    if(NOT MKL_${_mkl_component_upper}_LIBRARY)
      message(FATAL_ERROR
        "MKL library mkl_${_mkl_component} not found below ${_mkl_root}/lib or ${_mkl_root}/lib/intel64")
    endif()
    get_filename_component(_mkl_library_real
      "${MKL_${_mkl_component_upper}_LIBRARY}" REALPATH)
    _mkl_path_below_root("${_mkl_library_real}" "${_mkl_root}" _mkl_library_inside)
    if(NOT _mkl_library_inside)
      message(FATAL_ERROR
        "MKL library ${_mkl_component} resolves outside requested root ${_mkl_root}: ${_mkl_library_real}")
    endif()
    set(MKL_${_mkl_component_upper}_LIBRARY "${_mkl_library_real}" CACHE FILEPATH
      "Resolved MKL ${_mkl_component} library" FORCE)
  endforeach()
  message(STATUS "MKL selection: ${_mkl_selection}")
  message(STATUS "MKLROOT resolved: ${MKLROOT}")
  message(STATUS "MKL include: ${MKL_INCLUDE_DIR}")
  message(STATUS "MKL intel_lp64: ${MKL_INTEL_LP64_LIBRARY}")
  message(STATUS "MKL sequential: ${MKL_SEQUENTIAL_LIBRARY}")
  message(STATUS "MKL core: ${MKL_CORE_LIBRARY}")
  set(MKL_LIBRARIES
    "${MKL_INTEL_LP64_LIBRARY}"
    "${MKL_SEQUENTIAL_LIBRARY}"
    "${MKL_CORE_LIBRARY}")
endif()

function(add_density_target name source)
  add_executable(${name} ${source} src/skewMatUtils.cpp)
  target_include_directories(${name} PRIVATE inc "${EIGEN3_INCLUDE_DIR}" "${BOOST_INCLUDE_DIR}")
  target_compile_definitions(${name} PRIVATE PFQMC_SCALE_SAFE_UDT)
  target_compile_options(${name} PRIVATE -O3)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_link_options(${name} PRIVATE -no-pie)
    target_compile_options(${name} PRIVATE -fno-fast-math -ffp-contract=off)
    target_include_directories(${name} PRIVATE "${MKL_INCLUDE_DIR}")
    target_link_libraries(${name} PRIVATE ${MKL_LIBRARIES} pthread m dl)
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "Intel")
    target_compile_options(${name} PRIVATE -mkl=sequential -fp-model=precise -fimf-arch-consistency=true -no-fma)
    target_link_options(${name} PRIVATE -mkl=sequential)
  else()
    target_compile_options(${name} PRIVATE -Wno-enum-constexpr-conversion -qmkl=sequential -fp-model=precise -ffp-contract=off)
    target_link_options(${name} PRIVATE -qmkl=sequential)
  endif()
  target_link_libraries(${name} PRIVATE pf_c pf_fortran)
endfunction()
